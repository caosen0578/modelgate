## 1. 建表 SQL（强制执行，不得修改表名和字段名）

```sql
-- 模型实例
CREATE TABLE mg_model (
    id            BIGINT       NOT NULL AUTO_INCREMENT PRIMARY KEY,
    model_id      VARCHAR(64)  NOT NULL UNIQUE COMMENT '对外暴露的模型标识',
    model_name    VARCHAR(128) NOT NULL COMMENT '展示名称',
    provider      VARCHAR(32)  NOT NULL COMMENT 'openai|azure_openai|anthropic|private|custom',
    model_type    VARCHAR(32)  NOT NULL DEFAULT 'chat' COMMENT 'chat|embedding|image|audio',
    api_spec      VARCHAR(32)  NOT NULL DEFAULT 'openai' COMMENT 'openai|azure_openai|anthropic|custom',
    upstream_url  VARCHAR(512) NOT NULL COMMENT '上游请求地址',
    timeout_ms    INT          NOT NULL DEFAULT 30000 COMMENT '超时毫秒',
    default_params TEXT                 COMMENT '默认参数 JSON，业务可覆盖',
    force_params  TEXT                  COMMENT '强制参数 JSON，业务不可覆盖',
    param_mapping TEXT                  COMMENT '自定义字段映射 JSON，仅 api_spec=custom 时用',
    description   VARCHAR(512),
    enabled       TINYINT(1)   NOT NULL DEFAULT 1,
    created_at    DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at    DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 模型上游鉴权凭证（与 mg_model 一对一）
CREATE TABLE mg_model_credential (
    id          BIGINT      NOT NULL AUTO_INCREMENT PRIMARY KEY,
    model_id    VARCHAR(64) NOT NULL UNIQUE,
    auth_type   VARCHAR(32) NOT NULL COMMENT 'api_key|ak_sk|azure|none',
    access_key  VARCHAR(256)         COMMENT 'AK 或 Azure resource name，明文',
    secret_key  VARCHAR(512)         COMMENT '加密存储（AES-256-GCM）',
    extra_config TEXT                COMMENT 'Azure deployment_id/api_version 等，JSON',
    created_at  DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at  DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (model_id) REFERENCES mg_model(model_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 接入系统
CREATE TABLE mg_app (
    id         BIGINT       NOT NULL AUTO_INCREMENT PRIMARY KEY,
    app_id     VARCHAR(64)  NOT NULL UNIQUE COMMENT '应用唯一标识',
    app_name   VARCHAR(128) NOT NULL COMMENT '应用名称',
    sys_code   VARCHAR(64)  NOT NULL COMMENT '所属系统编码（如 CRM、BI）',
    app_code   VARCHAR(64)  NOT NULL COMMENT '应用编码（系统内唯一）',
    owner      VARCHAR(64)  NOT NULL COMMENT '负责人',
    remark     VARCHAR(512),
    enabled    TINYINT(1)   NOT NULL DEFAULT 1,
    created_at DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 接入系统凭证
CREATE TABLE mg_app_credential (
    id           BIGINT       NOT NULL AUTO_INCREMENT PRIMARY KEY,
    app_id       VARCHAR(64)  NOT NULL,
    cred_name    VARCHAR(128) NOT NULL COMMENT '凭证名称',
    cred_type    VARCHAR(32)  NOT NULL COMMENT 'api_key|ak_sk',
    access_key   VARCHAR(128)          COMMENT 'AK 明文 / API Key 末4位',
    secret_hash  VARCHAR(128)          COMMENT 'SHA-256(api_key) 或 SHA-256(sk)，用于校验',
    secret_encrypted VARCHAR(512)      COMMENT '加密后的 SK，仅 ak_sk 类型使用',
    enabled      TINYINT(1)   NOT NULL DEFAULT 1,
    expires_at   DATETIME              COMMENT 'NULL=永久',
    remark       VARCHAR(512),
    created_at   DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at   DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (app_id) REFERENCES mg_app(app_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 接入系统对模型的授权
CREATE TABLE mg_app_model_auth (
    id             BIGINT      NOT NULL AUTO_INCREMENT PRIMARY KEY,
    app_id         VARCHAR(64) NOT NULL,
    model_id       VARCHAR(64) NOT NULL,
    enabled        TINYINT(1)  NOT NULL DEFAULT 1,
    qps_limit      INT         NOT NULL DEFAULT 0 COMMENT '0=不限',
    daily_limit    INT         NOT NULL DEFAULT 0 COMMENT '0=不限',
    monthly_tokens BIGINT      NOT NULL DEFAULT 0 COMMENT '0=不限，单位 token',
    expires_at     DATETIME             COMMENT 'NULL=永久',
    created_at     DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at     DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY uk_app_model (app_id, model_id),
    FOREIGN KEY (app_id) REFERENCES mg_app(app_id),
    FOREIGN KEY (model_id) REFERENCES mg_model(model_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 访问控制规则
CREATE TABLE mg_access_rule (
    id         BIGINT       NOT NULL AUTO_INCREMENT PRIMARY KEY,
    app_id     VARCHAR(64)  NOT NULL,
    rule_type  VARCHAR(32)  NOT NULL COMMENT 'ip|cidr|domain|domain_wildcard',
    rule_value VARCHAR(256) NOT NULL COMMENT 'IP地址、CIDR、域名',
    remark     VARCHAR(256),
    created_at DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (app_id) REFERENCES mg_app(app_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 网关全局配置（保留原有表）
CREATE TABLE mg_gateway_config (
    config_key   VARCHAR(128) NOT NULL PRIMARY KEY,
    config_value TEXT         NOT NULL,
    description  VARCHAR(256),
    updated_by   VARCHAR(64),
    updated_at   DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 审计日志
CREATE TABLE mg_audit_log (
    id           BIGINT       NOT NULL AUTO_INCREMENT PRIMARY KEY,
    trace_id     VARCHAR(64)  NOT NULL,
    app_id       VARCHAR(64),
    model_id     VARCHAR(64),
    cred_type    VARCHAR(32),
    client_ip    VARCHAR(64),
    request_path VARCHAR(512),
    http_status  INT,
    latency_ms   INT,
    input_tokens INT          DEFAULT 0,
    output_tokens INT         DEFAULT 0,
    error_msg    VARCHAR(1024),
    created_at   DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_trace_id (trace_id),
    INDEX idx_app_id (app_id),
    INDEX idx_created_at (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

## 2. 管理接口定义（强制，路径和字段名不得更改）

### 2.1 通用响应结构

```json
{
  "code": 0,
  "message": "success",
  "data": {},
  "traceId": "abc123def456",
  "timestamp": 1714038000000
}
```

- `code=0` 表示成功，非 0 表示业务错误
- HTTP 状态码与 `code` 分开，HTTP 4xx/5xx 代表协议层错误，`code!=0` 代表业务拒绝

---

### 2.2 模型管理接口

**GET /admin/models** - 获取模型列表

Response `data`:
```json
[{
  "id": 1,
  "modelId": "gpt-4o",
  "modelName": "GPT-4o",
  "provider": "openai",
  "modelType": "chat",
  "apiSpec": "openai",
  "upstreamUrl": "https://api.openai.com",
  "timeoutMs": 30000,
  "enabled": true,
  "createdAt": "2026-04-23T10:00:00"
}]
```

**POST /admin/models** - 创建模型

Request body:
```json
{
  "modelId": "gpt-4o",
  "modelName": "GPT-4o",
  "provider": "openai",
  "modelType": "chat",
  "apiSpec": "openai",
  "upstreamUrl": "https://api.openai.com",
  "timeoutMs": 30000,
  "defaultParams": {},
  "forceParams": {},
  "description": "OpenAI GPT-4o",
  "credential": {
    "authType": "api_key",
    "secretKey": "sk-xxx"
  }
}
```

**PUT /admin/models/{modelId}** - 更新模型（不可修改 modelId）

**PATCH /admin/models/{modelId}/enable** - 启用模型（无请求体）

**PATCH /admin/models/{modelId}/disable** - 禁用模型（无请求体）

**DELETE /admin/models/{modelId}** - 删除模型（存在活跃授权时返回 400）

**POST /admin/models/{modelId}/test** - 连通性测试

Response `data`:
```json
{
  "success": true,
  "latencyMs": 342,
  "message": "连通成功"
}
```

### 2.3 接入系统接口

**GET /admin/apps** - 获取系统列表

**POST /admin/apps** - 创建接入系统

Request body:
```json
{
  "appId": "crm-service",
  "appName": "CRM系统",
  "owner": "张三",
  "contact": "zhangsan@corp.com",
  "remark": "客户关系管理系统"
}
```

**GET /admin/apps/{appId}** - 获取系统详情

**PUT /admin/apps/{appId}** - 更新系统信息（不可修改 appId）

**PATCH /admin/apps/{appId}/enable** - 启用系统

**PATCH /admin/apps/{appId}/disable** - 禁用系统

**DELETE /admin/apps/{appId}** - 删除系统（级联删除凭证、授权、访问规则）

---

### 2.4 凭证管理接口

**GET /admin/apps/{appId}/credentials** - 获取凭证列表

Response `data`:
```json
[{
  "id": 1,
  "credName": "生产密钥",
  "credType": "api_key",
  "accessKey": "****...xyz4",
  "enabled": true,
  "expiresAt": null,
  "createdAt": "2026-04-23T10:00:00"
}]
```

**POST /admin/apps/{appId}/credentials** - 生成凭证

Request body:
```json
{
  "credName": "生产密钥",
  "credType": "api_key",
  "expiresAt": null,
  "remark": "用于生产环境"
}
```

Response `data`（**此接口响应包含明文，仅此一次**）:
```json
{
  "id": 1,
  "credType": "api_key",
  "apiKey": "ModelGate-crm-service-sk_live_abc123xyz456",
  "expiresAt": null
}
```

对于 `ak_sk` 类型：
```json
{
  "id": 1,
  "credType": "ak_sk",
  "accessKey": "AK_crm_abc123",
  "secretKey": "SK_crm_xyz789...",
  "expiresAt": null
}
```

**PATCH /admin/apps/{appId}/credentials/{id}/enable** - 启用凭证

**PATCH /admin/apps/{appId}/credentials/{id}/disable** - 禁用凭证

**DELETE /admin/apps/{appId}/credentials/{id}** - 删除凭证

---

### 2.5 模型授权接口

**GET /admin/apps/{appId}/auth** - 获取授权列表

Response `data`:
```json
[{
  "id": 1,
  "modelId": "gpt-4o",
  "modelName": "GPT-4o",
  "enabled": true,
  "qpsLimit": 10,
  "dailyLimit": 10000,
  "monthlyTokens": 5000000,
  "expiresAt": null

```

**POST /admin/apps/{appId}/auth** - 添加模型授权

Request body:
```json
{
  "modelIds": ["gpt-4o", "claude-3-5-sonnet"],
  "qpsLimit": 10,
  "dailyLimit": 10000,
  "monthlyTokens": 0,

```

**PUT /admin/apps/{appId}/auth/{modelId}** - 更新授权配置

**PATCH /admin/apps/{appId}/auth/{modelId}/enable** - 启用授权

**PATCH /admin/apps/{appId}/auth/{modelId}/disable** - 禁用授权

**DELETE /admin/apps/{appId}/auth/{modelId}** - 撤销授权

### 2.6 访问控制接口

**GET /admin/apps/{appId}/access** - 获取访问规则列表

```json
  "id": 1,
```

### 2.7 网关配置接口

**GET /admin/config** - 获取所有配置

**GET /admin/config/{key}** - 获取单个配置

**PUT /admin/config/{key}** - 更新配置

Request body:
```json

```

## 3. 业务流量接口（对外兼容 OpenAI，不得更改）

```
POST /v1/chat/completions
Authorization: Bearer {api_key}
Content-Type: application/json

{
  "model": "{model_id}",
  "messages": [{"role": "user", "content": "hello"}],
  "stream": false
}
```

- 此路径由 Spring Cloud Gateway 直接路由，不经过 Controller
- AK/SK 鉴权时，Authorization 格式：`AKSK {access_key}:{timestamp}:{signature}`

```java
// ModelInstance.java
public class ModelInstance {
    private Long id;
    private String modelId;
    private String modelName;
    private String provider;
    private String modelType;
    private String apiSpec;

    private Integer timeoutMs;

    private String paramMapping;    // JSON string
    private String description;
    private Boolean enabled;
    private LocalDateTime createdAt;
    private LocalDateTime updatedAt;

// AppInfo.java
public class AppInfo {
    private Long id;
    private String appId;
    private String appName;

public class AppCredential {

public class AccessRule {
    private Long id;
```
