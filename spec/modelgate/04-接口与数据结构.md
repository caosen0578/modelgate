## 1. 建表 SQL（强制执行，不得修改表名和字段名）

```sql
-- 模型实例
CREATE TABLE mg_model (
    id            BIGINT       NOT NULL AUTO_INCREMENT PRIMARY KEY,
    model_id      VARCHAR(64)  NOT NULL UNIQUE COMMENT '对外暴露的模型标识',
    model_name    VARCHAR(128) NOT NULL COMMENT '展示名称',
    provider      VARCHAR(32)  NOT NULL COMMENT 'openai|azure_openai|anthropic|private|custom',
    model_type    VARCHAR(32)  NOT NULL DEFAULT 'chat' COMMENT 'chat|embedding|image|audio',
    api_spec      VARCHAR(32)  NOT NULL DEFAULT 'openai' COMMENT 'openai|azure_openai|anthropic|custom',
    upstream_url  VARCHAR(512) NOT NULL COMMENT '上游请求地址',
    timeout_ms    INT          NOT NULL DEFAULT 30000 COMMENT '超时毫秒',
    default_params TEXT                 COMMENT '默认参数 JSON，业务可覆盖',
    force_params  TEXT                  COMMENT '强制参数 JSON，业务不可覆盖',
    param_mapping TEXT                  COMMENT '自定义字段映射 JSON，仅 api_spec=custom 时用',
    description   VARCHAR(512),
    enabled       TINYINT(1)   NOT NULL DEFAULT 1,
    created_at    DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at    DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 模型上游鉴权凭证（与 mg_model 一对一）
CREATE TABLE mg_model_credential (
    id          BIGINT      NOT NULL AUTO_INCREMENT PRIMARY KEY,
    model_id    VARCHAR(64) NOT NULL UNIQUE,
    auth_type   VARCHAR(32) NOT NULL COMMENT 'api_key|ak_sk|azure|none',
    access_key  VARCHAR(256)         COMMENT 'AK 或 Azure resource name，明文',
    secret_key  VARCHAR(512)         COMMENT '加密存储（AES-256-GCM）',
    extra_config TEXT                COMMENT 'Azure deployment_id/api_version 等，JSON',
    created_at  DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at  DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (model_id) REFERENCES mg_model(model_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 接入系统
CREATE TABLE mg_app (
    id         BIGINT       NOT NULL AUTO_INCREMENT PRIMARY KEY,
    app_id     VARCHAR(64)  NOT NULL UNIQUE COMMENT '应用唯一标识',
    app_name   VARCHAR(128) NOT NULL COMMENT '应用名称',
    sys_code   VARCHAR(64)  NOT NULL COMMENT '所属系统编码（如 CRM、BI）',
    app_code   VARCHAR(64)  NOT NULL COMMENT '应用编码（系统内唯一）',
    owner      VARCHAR(64)  NOT NULL COMMENT '负责人',
    remark     VARCHAR(512),
    enabled    TINYINT(1)   NOT NULL DEFAULT 1,
    created_at DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 接入系统凭证
CREATE TABLE mg_app_credential (
    id           BIGINT       NOT NULL AUTO_INCREMENT PRIMARY KEY,
    app_id       VARCHAR(64)  NOT NULL,
    cred_name    VARCHAR(128) NOT NULL COMMENT '凭证名称',
    cred_type    VARCHAR(32)  NOT NULL COMMENT 'api_key|ak_sk',
    access_key   VARCHAR(128)          COMMENT 'AK 明文 / API Key 末4位',
    secret_hash  VARCHAR(128)          COMMENT 'SHA-256(api_key) 或 SHA-256(sk)，用于校验',
    secret_encrypted VARCHAR(512)      COMMENT '加密后的 SK，仅 ak_sk 类型使用',
    enabled      TINYINT(1)   NOT NULL DEFAULT 1,
    expires_at   DATETIME              COMMENT 'NULL=永久',
    remark       VARCHAR(512),
    created_at   DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at   DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (app_id) REFERENCES mg_app(app_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 接入系统对模型的授权
CREATE TABLE mg_app_model_auth (
    id             BIGINT      NOT NULL AUTO_INCREMENT PRIMARY KEY,
    app_id         VARCHAR(64) NOT NULL,
    model_id       VARCHAR(64) NOT NULL,
    enabled        TINYINT(1)  NOT NULL DEFAULT 1,
    qps_limit      INT         NOT NULL DEFAULT 0 COMMENT '0=不限',
    daily_limit    INT         NOT NULL DEFAULT 0 COMMENT '0=不限',
    monthly_tokens BIGINT      NOT NULL DEFAULT 0 COMMENT '0=不限，单位 token',
    expires_at     DATETIME             COMMENT 'NULL=永久',
    created_at     DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at     DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY uk_app_model (app_id, model_id),
    FOREIGN KEY (app_id) REFERENCES mg_app(app_id),
    FOREIGN KEY (model_id) REFERENCES mg_model(model_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 访问控制规则
CREATE TABLE mg_access_rule (
    id         BIGINT       NOT NULL AUTO_INCREMENT PRIMARY KEY,
    app_id     VARCHAR(64)  NOT NULL,
    rule_type  VARCHAR(32)  NOT NULL COMMENT 'ip|cidr|domain|domain_wildcard',
    rule_value VARCHAR(256) NOT NULL COMMENT 'IP地址、CIDR、域名',
    remark     VARCHAR(256),
    created_at DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (app_id) REFERENCES mg_app(app_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 网关全局配置（保留原有表）
CREATE TABLE mg_gateway_config (
    config_key   VARCHAR(128) NOT NULL PRIMARY KEY,
    config_value TEXT         NOT NULL,
    description  VARCHAR(256),
    updated_by   VARCHAR(64),
    updated_at   DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 审计日志
CREATE TABLE mg_audit_log (
    id             BIGINT       NOT NULL AUTO_INCREMENT PRIMARY KEY,
    trace_id       VARCHAR(64)  NOT NULL,
    app_id         VARCHAR(64),
    model_id       VARCHAR(64),
    cred_type      VARCHAR(32),
    client_ip      VARCHAR(64),
    caller_app_id  VARCHAR(64)           COMMENT '调用方系统标识（X-App-Id Header）',
    request_path   VARCHAR(512),
    http_status    INT,
    latency_ms     INT,
    input_tokens   INT          DEFAULT 0,
    output_tokens  INT          DEFAULT 0,
    error_msg      VARCHAR(1024),
    created_at     DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_trace_id (trace_id),
    INDEX idx_app_id (app_id),
    INDEX idx_created_at (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

## 2. 管理接口定义（强制，路径和字段名不得更改）

### 2.1 通用响应结构

```json
{
  "code": 0,
  "message": "success",
  "data": {},
  "traceId": "abc123def456",
  "timestamp": 1714038000000
}
```

- `code=0` 表示成功，非 0 表示业务错误
- HTTP 状态码与 `code` 分开，HTTP 4xx/5xx 代表协议层错误，`code!=0` 代表业务拒绝

### 2.2 模型管理接口

> **路径设计原则**：所有管理接口路径固定，变量通过 `@RequestParam` 或请求体传递，便于 RBAC 权限表按固定路径绑定角色。

**GET /admin/models** - 获取模型列表

Response `data`:
```json
[{
  "id": 1,
  "modelId": "gpt-4o",
  "modelName": "GPT-4o",
  "provider": "openai",
  "modelType": "chat",
  "apiSpec": "openai",
  "upstreamUrl": "https://api.openai.com",
  "timeoutMs": 30000,
  "enabled": true,
  "createdAt": "2026-04-23T10:00:00"
}]
```

**GET /admin/models/detail?modelId={modelId}** - 获取模型详情

**POST /admin/models** - 创建模型

Request body:
```json
{
  "modelId": "gpt-4o",
  "modelName": "GPT-4o",
  "provider": "openai",
  "modelType": "chat",
  "apiSpec": "openai",
  "upstreamUrl": "https://api.openai.com",
  "timeoutMs": 30000,
  "defaultParams": {},
  "forceParams": {},
  "description": "OpenAI GPT-4o",
  "credential": {
    "authType": "api_key",
    "secretKey": "sk-xxx"
  }
}
```

**PUT /admin/models/update** - 更新模型（body 中含 modelId，不可修改）

Request body: 同创建，必须包含 `modelId`

**PATCH /admin/models/enable?modelId={modelId}** - 启用模型

**PATCH /admin/models/disable?modelId={modelId}** - 禁用模型

**DELETE /admin/models/delete?modelId={modelId}** - 删除模型（存在活跃授权时返回 400）

**POST /admin/models/test?modelId={modelId}** - 连通性测试

Response `data`:
```json
{
  "success": true,
  "latencyMs": 342,
  "message": "连通成功"
}
```

## 5. 数据模型速查

### 5.1 ER 关系图

```
mg_model ─────────────────── mg_model_credential
   │ 1                                        1
   │
   │ N（通过 model_id 路由）
   │
mg_app_model_auth ──────────── mg_app
   N                                   │ 1
                                       │
                             ┌─────────┴──────────┐
                             │                    │
                     mg_app_credential   mg_access_rule
                        （凭证: N）            （访问规则: N）

mg_audit_log（独立，按 app_id/model_id 关联，无外键约束）
```

### 5.2 枚举值速查

**模型提供商 `provider`**

| 值 | 含义 |
|---|---|
| `openai` | OpenAI |
| `azure_openai` | Azure OpenAI |
| `anthropic` | Anthropic Claude |
| `private` | 私有化部署（OpenAI 兼容） |
| `custom` | 自定义上游 |

**模型类型 `model_type`**

| 值 | 含义 |
|---|---|
| `chat` | 对话补全 |
| `embedding` | 文本嵌入 |
| `image` | 图像生成 |
| `audio` | 语音 |

**API 规范 `api_spec`**

| 值 | 参数转换行为 |
|---|---|
| `openai` | 直接透传，不转换 |
| `azure_openai` | URL / Header 适配 |
| `anthropic` | 请求体 + 响应体全量转换 |
| `custom` | 按 `param_mapping` 字段转换 |

**上游鉴权类型 `auth_type`**

| 值 | 说明 |
|---|---|
| `api_key` | `Authorization: Bearer {key}` |
| `ak_sk` | HMAC 签名 |
| `azure` | `api-key` Header + URL 参数 |
| `none` | 不鉴权（内网私有模型） |

**系统凭证类型 `cred_type`**

| 值 | 说明 |
|---|---|
| `api_key` | 单一密钥，请求头 Bearer |
| `ak_sk` | AK 公开 + SK 签名 |

**访问规则类型 `rule_type`**

| 值 | 示例 | 典型用途 |
|---|---|---|
| `ip` | `192.168.1.100` | 办公区固定 IP 终端 |
| `cidr` | `10.244.0.0/16` | K8s Pod 网段（Pod IP 动态，需用网段） |
| `domain` | `app.internal.com` | 精确域名，匹配请求 `Origin` 头 |
| `domain_wildcard` | `*.internal.com` | 单级通配（`a.b.com` 不匹配 `*.com`） |

> **规则为空 = 不限制来源**，适合纯 K8s 集群内部全信任调用

### 5.3 关键业务级联规则

1. **App 禁用** → 该 App 下所有凭证立即失效（缓存强制刷新）
2. **凭证禁用** → 仅该凭证失效，同系统其他凭证不受影响
3. **模型授权禁用** → 该 App 访问该模型返回 403，其他模型不受影响
4. **模型禁用** → 所有 App 访问该模型返回 503
5. **授权过期** → 过期后等同于禁用，返回 403，记录不自动删除
6. **App 删除** → 级联删除该 App 下所有凭证、授权、访问规则
